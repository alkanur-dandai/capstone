const [start, setStart] = useState<Pin | null>(null);
  const [end, setEnd] = useState<Pin | null>(null);

type Pin = { latitude: number; longitude: number };

function RoutingController({
  start,
  end,
  setStart,
  setEnd,
  setRoute,
}: {
  start: Pin | null;
  end: Pin | null;
  setStart: (v: Pin | null) => void;
  setEnd: (v: Pin | null) => void;
  setRoute: (v: RouteGeoJSON | null) => void;
}) {
  const { current: mapRef } = useMap();

usePinSetter({
  map: mapRef ? mapRef.getMap() : null, // ✅ FIX
  start,
  end,
  setStart,
  setEnd,
  clearRoute: () => setRoute(null),
});


  // fetch route when start + end is ready
  useEffect(() => {
    async function loadRoute() {
      if (!start || !end) return;

      try {
        const route = await fetchOSRMRoute(
          [start.longitude, start.latitude],
          [end.longitude, end.latitude]
        );
        setRoute(route);
      } catch (err) {
        console.log("Routing error:", err);
      }
    }

    loadRoute();
  }, [start, end, setRoute]);

  return null;
}


<RoutingController
        start={start}
        end={end}
        setStart={setStart}
        setEnd={setEnd}
        setRoute={setRoute}
      />

      <RoutePins start={start} end={end} />
